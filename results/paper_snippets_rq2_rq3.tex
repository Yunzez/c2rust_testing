\begin{table*}[t]
  \centering
  \caption{\textbf{RQ3 --- name-independent matching under renaming and
    idiomatic restructuring.} Recall/precision are micro~(pooled functions) /
    macro~(unweighted mean over programs); the matcher never sees names. In the
    \emph{renamed} regime the LLM renamed nearly every function, so name-equality
    (what FLOURINE/RustAssure name-pairing recovers) collapses to $.124/.086$,
    while the structural matcher recovers $.876/.961$ ($\sim$7--11$\times$). The
    two LLMs rename very differently (name-eq $.124\!\to\!.613$) yet the matcher
    recall is \emph{stable} ($.876$ vs $.867$ micro). The \texttt{SACTOR} rows use
    \emph{independent} ground truth (its emitted \texttt{function\_name\_map}):
    ``names kept'' is a robustness check (io-shape transformed, name-eq trivially
    $1.0$); ``mechanically-renamed'' rewrites the Rust names to \texttt{r\_\#\#\#\#}
    so name-eq $\to 0.0$ on a corpus we did not label --- the matcher still
    recovers $100\%$ by structure.}
  \label{tab:rq3-matcher}
  \begin{tabular}{l r r r r cc c}
    \toprule
    & & & & & \multicolumn{2}{c}{recall ($\mu$/M)} & prec. ($\mu$/M) \\
    \cmidrule(lr){6-7}
    regime & progs & scor. & match. & corr.
      & matcher & name-eq & matcher \\
    \midrule
    raw-LLM \texttt{gpt-5-mini} --- \emph{renamed}     & 10 & 89 & 90 & 78 & \textbf{.876/.961} & .124/.086 & .867/.958 \\
    raw-LLM \texttt{gpt-5-nano} --- \emph{renamed}     & 10 & 75 & 86 & 65 & \textbf{.867/.914} & .613/.502 & .756/.891 \\
    SACTOR idiomatic --- names kept, io-shape transf.  &  2 & 12 & 14 & 12 & 1.000/1.000 & 1.000/1.000 & .857/.854 \\
    \textbf{SACTOR mech.-renamed} --- indep.\ truth     &  2 & 12 & 14 & 12 & \textbf{1.000/1.000} & \textbf{.000/.000} & .857/.854 \\
    \bottomrule
  \end{tabular}
\end{table*}
