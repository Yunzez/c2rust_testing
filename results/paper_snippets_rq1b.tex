\begin{table}[t]
  \centering
  \caption{\textbf{RQ1b --- recall on injected UB-free semantic bugs} (the recall half
    of the RQ2 claim: the same pipeline that reports \emph{zero} false bugs on faithful
    c2rust output must miss \emph{none} of the real ones). We inject 28 UB-free
    mutations (no pointer/memory operators) into faithful c2rust translations of six
    RQ2-frame programs and re-run the gated differential pipeline. An
    \emph{independent mutation-validity oracle} --- an exhaustive structured input
    sweep over a separately linked C\,+\,mutated-Rust binary, not our fuzzer ---
    fixes the denominator first: 1 mutant is semantically equivalent
    (\texttt{>}$\to$\texttt{>=} at a saturation boundary where both branches return
    the same value) and is excluded. The oracle's divergence-density spectrum spans
    $0.02$--$1.0$, including a \emph{single-input} case (dropping the Rust-side
    \texttt{INT\_MIN}/$-1$ guard while C keeps it; the triggering input is
    full-UBSan-clean on C). All 27 valid mutants are detected as
    \texttt{UB\_FREE\_DIVERGENCE}, each with a decoded triggering input proven UB-free
    on the C side. Single-run 25\,s budget per mutant; a 5-mutant$\times$3-seed
    re-run detects 15/15 (time-to-detect median $0.21$\,s, max $1.13$\,s).}
  \label{tab:rq1b-recall}
  \begin{tabular}{l r r r r r}
    \toprule
    operator class & inj. & equiv. & valid & det. & recall \\
    \midrule
    operator replacement   & 14 & 0 & 14 & 14 & 14/14 \\
    constant perturbation  &  7 & 0 &  7 &  7 & 7/7 \\
    guard weakening        &  3 & 0 &  3 &  3 & 3/3 \\
    off-by-one             &  4 & 1 &  3 &  3 & 3/3 \\
    \midrule
    \textbf{total} & \textbf{28} & \textbf{1} & \textbf{27} & \textbf{27} & \textbf{100\%} \\
    \bottomrule
  \end{tabular}
