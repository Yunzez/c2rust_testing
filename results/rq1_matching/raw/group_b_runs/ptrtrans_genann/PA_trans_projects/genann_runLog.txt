use rand::Rng;

pub fn genann_randomize(ann: Option<&mut Genann>) {
    // If ann is None (null in C), do nothing, matching the C behavior.
    if let Some(ann) = ann {
        // In C: for (i = 0; i < ann->total_weights; ++i)
        // Here we assume `total_weights` matches `weight.len()` logically.
        let total_weights = ann.total_weights.max(0) as usize;

        // Iterate over indices up to total_weights (or vec length, whichever is smaller)
        let len = ann.weight.len().min(total_weights);
        let mut rng = rand::thread_rng();

        for i in 0..len {
            // C: double r = (((double)rand())/2147483647);
            //     ann->weight[i] = r - 0.5;
            //
            // `rand()` in C returns [0, 2147483647]; rand::Rng::gen_range
            // generates in [0.0, 2147483647.0), which is a close analogue.
            let r: f64 = rng.gen_range(0.0..2147483647.0) / 2147483647.0;
            ann.weight[i] = r - 0.5;
        }
    }
}
